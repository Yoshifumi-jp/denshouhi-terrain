   ```powershell
   foreach ($i in 1..47) { $c = "{0:D2}" -f $i; Write-Output "=== $c ==="; python src/make_map.py --pref $c; if ($LASTEXITCODE -ne 0) { Write-Output "失敗：$c"; break } }
   ```
2. 県別ページを47都道府県分作り直す：
   ```powershell
   python src/build_site.py --pref all
   ```
